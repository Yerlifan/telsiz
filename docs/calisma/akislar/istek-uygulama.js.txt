function fixPrompt (id, wt, impl, rev) {
  const g = GRUPLAR[id]
  return `Telsiz deposunda plan grubu ${id} (${g.ad}) uygulandı ve bağımsız bir inceleyici aşağıdaki bulguları bildirdi. Çalışma kopyası: ${wt}. Grubun değişikliği: "git -C ${wt} diff ${impl.baslangic}..HEAD".
Bulgular:
${JSON.stringify(rev.bulgular, null, 1)}
İnceleyicinin test sonuçları: ${rev.testler}

Her bulguyu kodda doğrula. Gerçekse düzelt, gerekiyorsa test ekle. Gerçek değilse neden olmadığını kanıtıyla yaz. Düzeltmeleri "${id}: inceleme düzeltmeleri" başlıklı bir ya da birkaç commit olarak at.

${KURALLAR}`
}

async function grup (id, wt, zincir, hazirlik) {
  const impl = await agent(implPrompt(id, wt, hazirlik), { label: id + ' uygula', phase: zincir, schema: IMPL_SCHEMA })
  if (!impl) return { id, hata: 'uygulama ajanı sonuç vermedi' }
  log(id + ' uygulandı, tamam=' + impl.tamam + ', ' + impl.commitler.length + ' commit')
  const rev = await agent(reviewPrompt(id, wt, impl), { label: id + ' incele', phase: zincir, schema: REVIEW_SCHEMA })
  let fix = null
  if (rev && rev.bulgular.length > 0) {
    fix = await agent(fixPrompt(id, wt, impl, rev), { label: id + ' düzelt', phase: zincir, schema: FIX_SCHEMA })
  }
  log(id + ' bitti, bulgu ' + (rev ? rev.bulgular.length : 'yok') + ', düzeltilen ' + (fix ? fix.duzeltilen.length : 0))
  return { id, impl, inceleme: rev, duzeltme: fix }
}

async function zincir (ad, wt, ids, ilkHazirlik) {
  const sonuc = []
  for (let i = 0; i < ids.length; i++) {
    sonuc.push(await grup(ids[i], wt, ad, i === 0 ? ilkHazirlik : ''))
  }
  return sonuc
}
